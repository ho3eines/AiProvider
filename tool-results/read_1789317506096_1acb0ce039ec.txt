   464→function highlightCode(code, lang) {
   465→  var hashComment = HASH_COMMENT_LANGS.test(lang.trim());
   466→  var commentPart = hashComment
   467→    ? '(#[^\\n]*|\\/\\*[\\s\\S]*?\\*\\/|\\/\\/[^\\n]*)'

   470→    commentPart +
   471→      "|('(?:\\\\.|[^'\\\\\\n])*'|\"(?:\\\\.|[^\"\\\\\\n])*\"|\x60(?:\\\\.|[^\x60\\\\])*\x60)" +
   472→      '|\\b(\\d+(?:\\.\\d+)?)\\b' +
   473→      '|\\b(' + KEYWORDS.join('|') + ')\\b',
   474→    'g'
   475→  );

   477→  var last = 0;
   478→  var m;
   479→  while ((m = re.exec(code)) !== null) {
   480→    out += escapeHtml(code.slice(last, m.index));
   481→    if (m[1]) out += '<span class="tok-comment">' + escapeHtml(m[1]) + '</span>';
   482→    else if (m[2]) out += '<span class="tok-string">' + escapeHtml(m[2]) + '</span>';
   483→    else if (m[3]) out += '<span class="tok-number">' + escapeHtml(m[3]) + '</span>';
   484→    else out += '<span class="tok-keyword">' + escapeHtml(m[4]) + '</span>';
   485→    last = m.index + m[0].length;
   486→  }
   487→  out += escapeHtml(code.slice(last));
   488→  return out;

   506→function inlineFormat(s) {

   508→  // لینک [متن](https://…) — فقط http/https برای امنیت
   509→  out = out.replace(
   510→    /\[([^\]]+)\]\((https?:\/\/[^\s)]+)\)/g,
   511→    '<a href="$2" target="_blank" rel="noopener noreferrer" class="md-link">$1</a>'
   512→  );
   513→  // بولد **…**
   514→  out = out.replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>');
   515→  // ایتالیک *…*
   516→  out = out.replace(/(^|[^*])\*([^*\n]+)\*(?!\*)/g, '$1<em>$2</em>');
   517→  // خط‌خورده ~~…~~
   518→  out = out.replace(/~~([^~]+)~~/g, '<del>$1</del>');

   522→/**
   523→ * تبدیل مارک‌داون به HTML امن
   524→ * مراحل: استخراج بلوک کد → escape → کد اینلاین → پردازش بلاکی → بازگردانی placeholder ها
   525→ */
   526→function renderMarkdown(src) {
   527→  if (!src) return '';
   528→
   529→  /* ۱) استخراج بلوک‌های کد (حتی اگر در حین استریم هنوز بسته نشده باشند) */
   530→  var codeBlocks = [];
   531→  var text = src.replace(/\x60{3}([^\n\x60]*)\n?([\s\S]*?)(?:\x60{3}|$)/g, function (all, langLine, code) {
   532→    codeBlocks.push({ lang: (langLine || '').trim().split(/\s+/)[0] || '', code: code });
   533→    return '\u0000CB' + (codeBlocks.length - 1) + '\u0000';
   534→  });
   535→
   536→  /* ۲) escape کل متن — قبل از هر درج تگ */
   537→  text = escapeHtml(text);
   538→
   539→  /* ۳) کد اینلاین … → placeholder */
   540→  var inlineCodes = [];
   541→  text = text.replace(/\x60([^\x60\n]+)\x60/g, function (all, c) {
   542→    inlineCodes.push(c);
   543→    return '\u0000IC' + (inlineCodes.length - 1) + '\u0000';
   544→  });
   545→
   546→  /* ۴) پردازش بلاکی خط‌به‌خط */
   547→  var lines = text.split('\n');
   548→  var html = [];
   549→  var para = [];
   550→  var list = null; // {type:'ul'|'ol', items:[]}
   551→
   552→  function flushParagraph() {
   553→    if (!para.length) return;
   554→    html.push('<p class="md-p">' + para.map(inlineFormat).join('<br/>') + '</p>');
   555→    para = [];
   556→  }
   557→  function closeList() {
   558→    if (!list) return;
   559→    html.push(
   560→      '<' + list.type + ' class="md-list">' +
   561→      list.items.map(function (it) { return '<li>' + inlineFormat(it) + '</li>'; }).join('') +
   562→      '</' + list.type + '>'
   563→    );
   564→    list = null;
   565→  }
   566→
   567→  var i = 0;
   568→  while (i < lines.length) {
   569→    var trimmed = lines[i].trim();
   570→
   571→    // placeholder بلوک کد در خط مستقل
   572→    var cb = trimmed.match(/^\u0000CB(\d+)\u0000$/);
   573→    if (cb) {
   574→      flushParagraph();
   575→      closeList();
   576→      var b = codeBlocks[Number(cb[1])];
   577→      html.push(b ? codeBlockHtml(b.lang, b.code) : '');
   578→      i++;
   579→      continue;
   580→    }
   581→
   582→    // خط خالی
   583→    if (!trimmed) {
   584→      flushParagraph();
   585→      closeList();
   586→      i++;
   587→      continue;
   588→    }
   589→
   590→    // هدینگ # … ######
   591→    var h = trimmed.match(/^(#{1,6})\s+(.+)$/);
   592→    if (h) {
   593→      flushParagraph();
   594→      closeList();
   595→      var lv = h[1].length;
   596→      html.push('<h' + lv + ' class="md-h">' + inlineFormat(h[2]) + '</h' + lv + '>');
   597→      i++;
   598→      continue;
   599→    }
   600→
   601→    // خط افقی --- *** ___
   602→    if (/^(-{3,}|\*{3,}|_{3,})$/.test(trimmed)) {
   603→      flushParagraph();
   604→      closeList();
   605→      html.push('<hr class="md-hr"/>');
   606→      i++;
   607→      continue;
   608→    }
   609→
   610→    // نقل‌قول > (در متن escape شده به صورت &gt;)
   611→    if (/^(&gt;|>)\s?/.test(trimmed)) {
   612→      flushParagraph();
   613→      closeList();
   614→      var quote = [];
   615→      while (i < lines.length) {
   616→        var m2 = lines[i].trim().match(/^(&gt;|>)\s?(.*)$/);
   617→        if (!m2) break;
   618→        quote.push(m2[2]);
   619→        i++;
   620→      }
   621→      html.push('<blockquote class="md-quote">' + quote.map(inlineFormat).join('<br/>') + '</blockquote>');
   622→      continue;
   623→    }
   624→
   625→    // لیست نامرتب - * +
   626→    var ul = trimmed.match(/^[-*+]\s+(.+)$/);
   627→    if (ul) {
   628→      flushParagraph();
   629→      if (!list || list.type !== 'ul') {
   630→        closeList();
   631→        list = { type: 'ul', items: [] };
   632→      }
   633→      list.items.push(ul[1]);
   634→      i++;
   635→      continue;
   636→    }
   637→
   638→    // لیست مرتب 1. 2)
   639→    var ol = trimmed.match(/^(\d{1,3})[.)]\s+(.+)$/);
   640→    if (ol) {
   641→      flushParagraph();
   642→      if (!list || list.type !== 'ol') {
   643→        closeList();
   644→        list = { type: 'ol', items: [] };
   645→      }
   646→      list.items.push(ol[2]);
   647→      i++;
   648→      continue;
   649→    }
   650→
   651→    // پاراگراف
   652→    para.push(trimmed);
   653→    i++;
   654→  }
   655→  flushParagraph();
   656→  closeList();
   657→
   658→  /* ۵) بازگردانی placeholder ها */
   659→  var out = html.join('\n');
   660→  out = out.replace(/\u0000CB(\d+)\u0000/g, function (a, n) {
   661→    var bb = codeBlocks[Number(n)];
   662→    return bb ? codeBlockHtml(bb.lang, bb.code) : '';
   663→  });
   664→  out = out.replace(/\u0000IC(\d+)\u0000/g, function (a, n) {
   665→    return '<code class="md-icode">' + inlineCodes[Number(n)] + '</code>';
   666→  });
   667→  return out;
   668→}

   685→   - data: [DONE]                                       (پایان استریم)
   686→   فیلدهای تفکر: reasoning_content / reasoning / thinking / delta.thinking
   687→*/
   688→function createSseParser(opts) {
   689→  var buf = '';         // بافر خطوط ناقص
   690→  var pendingJson = ''; // JSON چندخطیِ ناقص
   691→
   692→  function emit(text, reasoning, error) {
   693→    text = text || '';
   694→    reasoning = reasoning || '';
   695→    if (text || reasoning || error) opts.onEvent({ text: text, reasoning: reasoning, error: error });
   696→  }
   697→
   698→  function addText(v, out) {
   699→    if (typeof v === 'string') out.text += v;
   700→    else if (Array.isArray(v)) {
   701→      for (var i = 0; i < v.length; i++) {
   702→        var b = v[i];
   703→        if (typeof b === 'string') out.text += b;
   704→        else if (b && typeof b === 'object' && typeof b.text === 'string') out.text += b.text;
   705→      }
   706→    }
   707→  }
   708→  function addReason(v, out) {
   709→    if (typeof v === 'string') out.reasoning += v;
   710→  }
   711→
   712→  function applyObject(obj) {
   713→    var out = { text: '', reasoning: '' };
   714→    var error;
   715→    var o = obj;
   716→    if (!o || typeof o !== 'object') {
   717→      emit(String(obj));
   718→      return;
   719→    }
   720→
   721→    // --- سبک OpenAI: choices[0].delta / choices[0].message ---
   722→    var choice = Array.isArray(o.choices) ? o.choices[0] : undefined;
   723→    if (choice && typeof choice === 'object') {
   724→      var delta = choice.delta;
   725→      var message = choice.message;
   726→      if (delta && typeof delta === 'object') {
   727→        addText(delta.content, out);
   728→        addText(delta.text, out);
   729→        addReason(delta.reasoning_content, out);
   730→        addReason(delta.reasoning, out);
   731→        addReason(delta.thinking, out);
   732→      }
   733→      if (message && typeof message === 'object') {
   734→        addText(message.content, out);
   735→        addText(message.text, out);
   736→        addReason(message.reasoning_content, out);
   737→        addReason(message.reasoning, out);
   738→        addReason(message.thinking, out);
   739→      }
   740→      if (typeof choice.text === 'string') addText(choice.text, out);
   741→    }
   742→
   743→    // --- سبک Claude: delta.text / delta.thinking ---
   744→    var topDelta = o.delta;
   745→    if (topDelta && typeof topDelta === 'object') {
   746→      addText(topDelta.text, out);
   747→      addText(topDelta.content, out);
   748→      addReason(topDelta.thinking, out);
   749→      addReason(topDelta.reasoning_content, out);
   750→      addReason(topDelta.reasoning, out);
   751→    }
   752→    var topMessage = o.message;
   753→    if (topMessage && typeof topMessage === 'object') {
   754→      addText(topMessage.content, out);
   755→      addReason(topMessage.reasoning_content, out);
   756→      addReason(topMessage.thinking, out);
   757→    }
   758→
   759→    // --- فیلدهای مستقیم روی ریشه ---
   760→    addText(o.content, out);
   761→    addText(o.text, out);
   762→    addReason(o.reasoning_content, out);
   763→    addReason(o.reasoning, out);
   764→    addReason(o.thinking, out);
   765→
   766→    // --- خطای داخل استریم ---
   767→    if (o.error) {
   768→      error = typeof o.error === 'string'
   769→        ? o.error
   770→        : (o.error && typeof o.error.message === 'string')
   771→          ? o.error.message
   772→          : JSON.stringify(o.error);
   773→    }
   774→
   775→    emit(out.text, out.reasoning, error);
   776→  }
   777→
   778→  function flushPending() {
   779→    if (!pendingJson) return;
   780→    emit(pendingJson); // JSON ناتمام → به‌صورت خام
   781→    pendingJson = '';
   782→  }
   783→
   784→  function handlePayload(payload) {
   785→    var t = payload.trim();
   786→    if (!t) return;
   787→    if (t === '[DONE]') return; // پایان استریم
   788→    if (t.charAt(0) === '{' || t.charAt(0) === '[') {
   789→      // JSON ممکن است چندخطی باشد؛ تا کامل شدن نگه می‌داریم
   790→      pendingJson = pendingJson ? pendingJson + '\n' + payload : payload;
   791→      try {
   792→        applyObject(JSON.parse(pendingJson));
   793→        pendingJson = '';
   794→      } catch (e) { /* هنوز ناقص است؛ خط بعدی ادامه می‌دهد */ }
   795→    } else {
   796→      // رشتهٔ خام غیر JSON → مستقیم به متن
   797→      flushPending();
   798→      emit(payload);
   799→    }
   800→  }
   801→
   802→  function processLine(line) {
   803→    if (line === '') { flushPending(); return; }
   804→    if (line.charAt(0) === ':') return; // کامنت SSE مثل ": ping"
   805→    if (line.indexOf('data:') === 0) {
   806→      var p = line.slice(5);
   807→      if (p.charAt(0) === ' ') p = p.slice(1);
   808→      handlePayload(p);
   809→      return;
   810→    }
   811→    if (/^(event|id|retry)\s*:/i.test(line)) return; // متادیتای SSE
   812→    handlePayload(line); // خط خام (مثلاً {"content":"..."} بدون data:)
   813→  }
   814→
   815→  return {
   816→    feed: function (chunk) {
   817→      buf += chunk;
   818→      var nl;
   819→      while ((nl = buf.indexOf('\n')) !== -1) {
   820→        var line = buf.slice(0, nl).replace(/\r$/, '');
   821→        buf = buf.slice(nl + 1);
   822→        processLine(line);
   823→      }
   824→    },
   825→    end: function () {
   826→      if (buf.trim()) {
   827→        processLine(buf.replace(/\r$/, ''));
   828→        buf = '';
   829→      }
   830→      if (pendingJson) {
   831→        emit(pendingJson);
   832→        pendingJson = '';
   833→      }
   834→    }
   835→  };
   836→}
