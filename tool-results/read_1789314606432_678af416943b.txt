    92→async function copyText(t: string): Promise<boolean> {
    93→  try {
    94→    await navigator.clipboard.writeText(t);
    95→    return true;
    96→  } catch {
    97→    try {
    98→      const ta = document.createElement('textarea');
    99→      ta.value = t;
   100→      ta.style.position = 'fixed';
   101→      ta.style.opacity = '0';
   102→      document.body.appendChild(ta);
   103→      ta.select();
   104→      const ok = document.execCommand('copy');
   105→      ta.remove();
   106→      return ok;
   107→    } catch {
   108→      return false;
   109→    }
   110→  }
   111→}

   113→function downloadFile(name: string, content: string, mime: string): void {
   114→  const blob = new Blob([content], { type: mime + ';charset=utf-8' });
   115→  const url = URL.createObjectURL(blob);
   116→  const a = document.createElement('a');
   117→  a.href = url;
   118→  a.download = name;
   119→  document.body.appendChild(a);
   120→  a.click();
   121→  a.remove();
   122→  setTimeout(() => URL.revokeObjectURL(url), 1500);
   123→}

   605→                updatedAt: Date.now(),

   721→        abortRef.current = null;
   722→        setLiveMsgId(null);
   723→        setStreaming(false);
   724→      }
